    /// Vertical Hack Squat: the shoulders coming forward and down off the
    /// pads, the trunk folding forward out of the carriage. The trunk and the
    /// shoulders tip `degrees` forward about the pelvis. Drawn without the
    /// arms: the hands hold the carriage's handles beside the head, and swung
    /// with the trunk, as `leanedForward` draws them, they crossed it in a
    /// zigzag from the framing. At the bottom (2.0 s, knees 88°) 25° carries
    /// the shoulders ~19 cm forward and ~4 cm down, the head ~26 cm forward.
    /// Grows with the knees.
    private static func female1109TrunkFolded(_ degrees: Float) -> FaultPose {
        FaultPose(chains: [spine, shoulders],
                  moves: [.turn(pivot: "pelvis", points: torso + ["upper_arm_*"], axis: .lateral, degrees: -degrees)],
                  strength: .withBend("shin_L"))
    }

    /// Vertical Hack Squat: both feet's toes turned `degrees` out about the
    /// ankles, the heels left where they are (the model's feet point ~7°
    /// out). A setup fault, the same all rep: 40° swings each toe tip ~13 cm
    /// (~11.5 cm out to the side, ~6 cm back).
    private static func female1109ToesOut(_ degrees: Float) -> FaultPose {
        FaultPose(chains: [legs],
                  moves: [.turn(pivot: "foot_*", points: ["foot_*.tip"], axis: .up, degrees: -degrees)])
    }
