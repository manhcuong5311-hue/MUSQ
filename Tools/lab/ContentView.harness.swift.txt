//
//  ContentView.swift — TEMP HARNESS (lab mirrors only, never commit)
//
//  HARNESS_VIEW / HARNESS_PHOTO: the viewport alone (382x655 trainer size, or
//  the 380x380 white thumbnail), with optional HARNESS_FRAMING
//  "yaw,zoom,x,y,z" and HARNESS_STILL seconds. HARNESS_EXERCISE: the trainer.
//  HARNESS_FAULT + HARNESS_CUE: the trainer opened on that cue's mistake
//  (launch with DEBUG_PREMIUM=1). Exercise3DView reads Purchases, Ads and
//  Paywall from the environment, so the harness injects them.
//  HARNESS_RESOURCE (2026-10-10): with HARNESS_VIEW/PHOTO, load that model
//  resource instead of the exercise's own (e.g. one stance of a stance set).
//  HARNESS_XRAY (2026-10-21): an x-ray demo's viewport by resource name, at
//  HARNESS_XRAY_VIEW (an XRayView raw value, default auto), paused at
//  HARNESS_STILL seconds once the model has loaded (shoot after ~8 s).
//  HARNESS_MAP (2026-10-22): the trainer's Muscles Worked panel for an
//  exercise ("Name" or "Name|Stance"), with its model's chest paint.
//  HARNESS_WEEK: the Muscles tab's map for a made-up week, "Name:sets,...".
//

import SwiftUI

struct ContentView: View {
    var body: some View {
        let env = ProcessInfo.processInfo.environment
        let still = env["HARNESS_STILL"].flatMap { TimeInterval($0) }
        if let name = env["HARNESS_VIEW"] ?? env["HARNESS_PHOTO"],
           let ex = SampleData.exercises.first(where: { $0.name == name }),
           let m = SampleData.model(for: ex) {
            let photo = env["HARNESS_PHOTO"] != nil
            let f = env["HARNESS_FRAMING"]?.split(separator: ",").compactMap { Float($0) } ?? []
            let framing = f.count == 5
                ? ModelFraming(yaw: f[0], zoom: f[1], offset: [f[2], f[3], f[4]]) : m.framing
            let resource = env["HARNESS_RESOURCE"] ?? m.resource
            ZStack {
                (photo ? Color.white : Color.black).ignoresSafeArea()
                if photo {
                    Viewport(slot: "", model: resource, framing: framing, speed: m.speed,
                             still: still, inner: .white, outer: .white, cornerRadius: 0) { EmptyView() }
                        .frame(width: 380, height: 380)
                } else {
                    Viewport(slot: "", model: resource, framing: framing, speed: m.speed,
                             still: still) { EmptyView() }
                        .frame(width: 382, height: 655)
                }
            }
        } else if let name = env["HARNESS_FAULT"],
                  let ex = SampleData.exercises.first(where: { $0.name == name }) {
            let purchases = Purchases()
            NavigationStack {
                Exercise3DView(exercise: ex, cue: env["HARNESS_CUE"], showingMistake: true, still: still)
            }
            .environment(WorkoutStore())
            .environment(RestTimer())
            .environment(purchases)
            .environment(Ads(purchases: purchases))
            .environment(Paywall())
        } else if let name = env["HARNESS_EXERCISE"],
                  let ex = SampleData.exercises.first(where: { $0.name == name }) {
            let purchases = Purchases()
            NavigationStack { Exercise3DView(exercise: ex, still: still) }
                .environment(WorkoutStore())
                .environment(RestTimer())
                .environment(purchases)
                .environment(Ads(purchases: purchases))
                .environment(Paywall())
        } else if let spec = env["HARNESS_MAP"] {
            MapHarness(spec: spec)
        } else if let spec = env["HARNESS_WEEK"] {
            WeekHarness(spec: spec)
        } else if let res = env["HARNESS_XRAY"],
                  let demo = [XRayDemos.benchScapula, XRayDemos.backBarSetup, XRayDemos.deadliftSetup]
                      .first(where: { $0.resource == res }) {
            XRayHarness(demo: demo, view: XRayView(rawValue: env["HARNESS_XRAY_VIEW"] ?? "") ?? .auto, still: still)
        } else {
            RootView()
        }
    }
}

struct XRayHarness: View {
    let demo: XRayDemo
    let view: XRayView
    let still: TimeInterval?
    @State private var stage: XRayStage

    init(demo: XRayDemo, view: XRayView, still: TimeInterval?) {
        self.demo = demo; self.view = view; self.still = still
        _stage = State(initialValue: XRayStage(demo: demo))
    }

    var body: some View {
        // The demo screen's viewport on a phone: full width, between the
        // header and the controls.
        XRayViewport(stage: stage)
            .frame(width: 402, height: 560)
            .background(Color(white: 0.1))
            .task {
                stage.pick(view)
                try? await Task.sleep(for: .seconds(4))
                if let still {
                    stage.seek(to: still)
                    if !stage.paused { stage.togglePause() }
                }
            }
    }
}


struct MapHarness: View {
    let spec: String

    var body: some View {
        let parts = spec.split(separator: "|").map(String.init)
        if let ex = SampleData.exercises.first(where: { $0.name == parts[0] }),
           let content = SampleData.content(for: ex) {
            let stance = parts.count > 1 ? SampleData.stances(for: ex).first { $0.label == parts[1] } : nil
            let model = stance?.model ?? SampleData.model(for: ex)
            ZStack {
                DS.ink.ignoresSafeArea()
                MuscleActivationContent(exerciseName: ex.name.uppercased(),
                                        muscles: stance?.activation ?? content.activation,
                                        stabilisers: content.stabilisers,
                                        chest: model.flatMap { ChestPaint.heads(forModel: $0.resource) })
                    .padding(.horizontal, 20)
            }
        }
    }
}

struct WeekHarness: View {
    let spec: String

    var body: some View {
        let now = Date()
        let exercises: [WorkoutExercise] = spec.split(separator: ",").compactMap { item in
            let kv = item.split(separator: ":").map(String.init)
            guard kv.count == 2, let n = Int(kv[1]) else { return nil }
            return WorkoutExercise(
                exerciseName: kv[0], group: .chest, repRange: RepRange(8, 12),
                sets: (0..<n).map { _ in WorkoutSet(reps: 10, isCompleted: true, completedAt: now) },
                contributions: ExerciseCatalog.contributions(forExerciseNamed: kv[0]),
                chestHeads: ChestPaint.heads(forExerciseNamed: kv[0]))
        }
        let sessions = [WorkoutSession(day: Calendar.current.startOfDay(for: now), exercises: exercises)]
        let calc = MuscleActivityCalculator()
        var fills: [BodyRegion: Color] = [:]
        for entry in calc.activity(from: sessions, period: .week, now: now) {
            guard let color = entry.level.color else { continue }
            for region in entry.muscle.bodyRegions { fills[region] = color }
        }
        for (region, level) in zip(BodyRegion.chestHeads, calc.chest(from: sessions, period: .week, now: now).levels) {
            fills[region] = level.color
        }
        return ZStack {
            DS.ink.ignoresSafeArea()
            BodyMapCanvas(side: .front, fills: fills, lineWidth: 1)
                .frame(width: 360, height: 700)
        }
    }
}
